// ==UserScript==
// @name         Modernize & Dark Mode for 10.145.22.11
// @namespace    http://tampermonkey.net/
// @version      2.1
// @description  Replaces legacy .gif tree icons, applies a global dark theme, and highlights selected rows.
// @match        http://10.145.22.11:8888/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // ==========================================
    // 1. DARK MODE INJECTION
    // ==========================================
    function applyDarkMode() {
        const css = `
            /* Global Background and Text */
            html, body {
                background-color: #121212 !important;
                color: #e0e0e0 !important;
            }

            /* Links */
            a {
                color: #66b3ff !important;
                text-decoration: none !important;
            }
            a:hover {
                text-decoration: underline !important;
                color: #99ccff !important;
            }

            /* Tables, Rows, and Cells */
            table, th, td, tr, tbody, thead {
                border-color: #444 !important;
                background-color: #1e1e1e !important;
                color: #e0e0e0 !important;
            }
            
            /* Table Headers */
            th {
                background-color: #2c2c2c !important;
                font-weight: bold !important;
            }

            /* ===== NEW: Highlight Selected Rows ===== */
            /* This highlights the entire row if a checkbox inside it is checked */
            tr:has(input[type="checkbox"]:checked),
            tr:has(input[type="checkbox"]:checked) td {
                background-color: #003b5c !important; /* A visible dark blue highlight */
                color: #ffffff !important;
            }

            /* Make Checkboxes More Visible */
            input[type="checkbox"] {
                transform: scale(1.3) !important; /* Make them slightly larger */
                accent-color: #4CAF50 !important; /* Green background when checked */
                cursor: pointer;
                margin: 4px !important;
            }

            /* Inputs, Textareas, Buttons, and Dropdowns */
            input, select, textarea, button {
                background-color: #333 !important;
                color: #fff !important;
                border: 1px solid #555 !important;
                border-radius: 4px !important;
                padding: 3px 6px !important;
                color-scheme: dark !important;
            }
            
            button, input[type="button"], input[type="submit"], input[type="reset"] {
                cursor: pointer !important;
                background-color: #4CAF50 !important;
                border: none !important;
                font-weight: bold !important;
            }
            
            button:hover, input[type="button"]:hover, input[type="submit"]:hover {
                background-color: #45a049 !important;
            }

            /* Divs and Containers */
            div, span, p, label {
                background-color: transparent !important;
                color: inherit !important;
            }

            /* Specifically overriding inline backgrounds */
            [bgcolor], [style*="background"] {
                background-color: #1e1e1e !important;
                background-image: none !important;
            }
        `;

        const styleNode = document.createElement('style');
        styleNode.type = 'text/css';
        styleNode.appendChild(document.createTextNode(css));
        document.head.appendChild(styleNode);
    }

    applyDarkMode();

    // ==========================================
    // 2. ICON REPLACEMENT
    // ==========================================
    const iconMap = {
        'folderopen.gif': 'https://img.icons8.com/fluency/48/opened-folder.png',
        'folder.gif': 'https://img.icons8.com/fluency/48/folder-invoices.png',
        'page.gif': 'https://img.icons8.com/fluency/48/document.png',
        'base.gif': 'https://img.icons8.com/fluency/48/domain.png'
    };

    function replaceIcons(node) {
        const images = (node.tagName && node.tagName.toLowerCase() === 'img') 
            ? [node] 
            : (node.querySelectorAll ? node.querySelectorAll('img') : []);

        images.forEach(img => {
            if (img.src) {
                for (const [oldIcon, newIcon] of Object.entries(iconMap)) {
                    if (img.src.endsWith('/img/' + oldIcon)) {
                        img.src = newIcon;
                        img.style.width = '18px';
                        img.style.height = '18px';
                        img.style.objectFit = 'contain';
                    }
                }
            }
        });
    }

    replaceIcons(document.body);

    const observer = new MutationObserver(mutations => {
        mutations.forEach(mutation => {
            if (mutation.type === 'childList') {
                mutation.addedNodes.forEach(node => {
                    if (node.nodeType === 1) replaceIcons(node);
                });
            } else if (mutation.type === 'attributes' && mutation.attributeName === 'src') {
                if (mutation.target.tagName.toLowerCase() === 'img') {
                    if (mutation.target.src.includes('/img/')) {
                         replaceIcons(mutation.target);
                    }
                }
            }
        });
    });

    observer.observe(document.body, {
        childList: true,
        subtree: true,
        attributes: true,
        attributeFilter: ['src']
    });

})();
